# DSRL Steering of Frozen GR00T N1.7 — Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Train a small SAC agent whose action is GR00T N1.7's flow-matching initial noise, improving colored-cube sort reliability on a Unitree G1 in MuJoCo without touching GR00T's weights.

**Architecture:** GR00T stays frozen. Its backbone pass is separated from its flow ODE, so a SAC actor reads pooled backbone features (already computed, free) and emits the noise seed that the ODE integrates into an action chunk. Learning uses DSRL-NA: a twin action-space critic `Q^a` carries the TD bootstrap, a twin noise-space critic `Q^w` is distilled from it, and the bootstrap is routed through `Q^w` so no GR00T forward pass ever appears in the update path.

**Tech Stack:** Python 3.10, PyTorch 2.7.1+cu128 (CUDA 12.8, sm_120), MuJoCo, NumPy, pytest.

```python
"""Plain data describing a sort scene. No MuJoCo, no torch — deliberately."""

from dataclasses import dataclass, field

Vec3 = tuple[float, float, float]


@dataclass(frozen=True)
class CubeState:
    position: Vec3
    color: str
    speed: float  # linear speed magnitude, m/s
    grasped: bool


@dataclass(frozen=True)
class TargetState:
    color: str
    center: Vec3
    half_extent: Vec3
    up_axis: Vec3 = (0.0, 0.0, 1.0)


@dataclass(frozen=True)
class SortState:
    cubes: tuple[CubeState, ...]
    targets: tuple[TargetState, ...]
    active_index: int
    settle_counts: tuple[int, ...]
    placed_mask: tuple[bool, ...]
    pelvis_height: float
    chunk_index: int

    @property
    def active_cube(self) -> CubeState:
        return self.cubes[self.active_index]

    def target_for_color(self, color: str) -> TargetState | None:
        for target in self.targets:
            if target.color == color:
                return target
        return None


@dataclass(frozen=True)
class RewardResult:
    reward: float
    terms: dict[str, float] = field(default_factory=dict)
    terminal: bool = False
    reason: str = ""
```

```python
"""Reward as pure functions over scene state.

Deliberately free of MuJoCo and torch so every term is unit-testable in
milliseconds with hand-written poses.
"""

import math

from dsrl.config import RewardConfig
from dsrl.sim.types import CubeState, RewardResult, SortState, TargetState


def is_contained(cube: CubeState, target: TargetState) -> bool:
    """True when the cube centre lies inside the target volume (inclusive)."""
    return all(
        abs(cube.position[axis] - target.center[axis]) <= target.half_extent[axis]
        for axis in range(3)
    )


def _containing_target(cube: CubeState, state: SortState) -> TargetState | None:
    for target in state.targets:
        if is_contained(cube, target):
            return target
    return None


def advance_settle_counts(state: SortState, cfg: RewardConfig) -> tuple[int, ...]:
    """Increment per-cube settle counters; reset any cube that is moving.

    The speed clause is load-bearing: without it a cube tumbling through a
    target would score, and the policy would learn to throw.
    """
    counts = []
    for index, cube in enumerate(state.cubes):
        contained = _containing_target(cube, state) is not None
        if contained and cube.speed < cfg.settle_speed:
            counts.append(state.settle_counts[index] + 1)
        else:
            counts.append(0)
    return tuple(counts)


def is_settled(count: int, cfg: RewardConfig) -> bool:
    return count >= cfg.settle_steps


def _tipped(target: TargetState, cfg: RewardConfig) -> bool:
    dot = target.up_axis[2] / (math.sqrt(sum(a * a for a in target.up_axis)) or 1.0)
    dot = max(-1.0, min(1.0, dot))
    return math.degrees(math.acos(dot)) > cfg.tip_angle_deg


def _distance(a: tuple[float, ...], b: tuple[float, ...]) -> float:
    return math.sqrt(sum((x - y) ** 2 for x, y in zip(a, b)))


def compute_reward(
    state: SortState,
    prev: SortState,
    chunk_delta: float,
    cfg: RewardConfig,
) -> RewardResult:
    """Assemble the reward for one chunk transition.

    `chunk_delta` is the norm of the difference between this action chunk and
    the previous one, used only for the jerk regulariser.
    """
    terms: dict[str, float] = {}
    terminal = False
    reason = ""

    # --- robot fell: dominates everything ---
    if state.pelvis_height < cfg.fall_height:
        terms["fell"] = cfg.fell
        return RewardResult(
            reward=sum(terms.values()), terms=terms, terminal=True, reason="fell"
        )

    active = state.active_cube
    active_target = _containing_target(active, state)
    settled = is_settled(state.settle_counts[state.active_index], cfg)

    # --- terminal placement outcomes ---
    if settled and active_target is not None:
        if active_target.color == active.color:
            terms["correct_placement"] = cfg.correct_placement
            terminal, reason = True, "correct_placement"
        else:
            terms["wrong_target"] = cfg.wrong_target
            terminal, reason = True, "wrong_target"

    # --- dropped: at rest, ungrasped, outside every target ---
    elif (
        active_target is None
        and not active.grasped
        and active.speed < cfg.settle_speed
        and state.settle_counts[state.active_index] == 0
        and prev.settle_counts[state.active_index] == 0
        and state.chunk_index > 0
        and _distance(active.position, prev.active_cube.position) < 1e-3
    ):
        terms["dropped"] = cfg.dropped
        terminal, reason = True, "dropped"

    # --- disturbing already-placed cubes ---
    disturbed = 0
    for index, was_placed in enumerate(prev.placed_mask):
        if not was_placed or index == state.active_index:
            continue
        if _containing_target(state.cubes[index], state) is None:
            disturbed += 1
    if disturbed:
        terms["disturb"] = cfg.disturb * disturbed

    # --- tipped receptacles ---
    tipped = sum(1 for target in state.targets if _tipped(target, cfg))
    if tipped:
        terms["target_tipped"] = cfg.target_tipped * tipped

    # --- grasp slip: was held, now free and moving, not settled ---
    if prev.active_cube.grasped and not active.grasped and not settled:
        if active.speed >= cfg.settle_speed:
            terms["grasp_slip"] = cfg.grasp_slip

    # --- jerk regulariser (dense but tiny; not shaping) ---
    if chunk_delta:
        terms["jerk"] = cfg.jerk_coeff * chunk_delta

    # --- optional approach shaping, disabled by default ---
    if cfg.approach_coeff:
        goal = state.target_for_color(active.color)
        if goal is not None:
            progress = _distance(prev.active_cube.position, goal.center) - _distance(
                active.position, goal.center
            )
            terms["approach"] = min(cfg.approach_coeff * progress, cfg.approach_cap)

    return RewardResult(
        reward=sum(terms.values()), terms=terms, terminal=terminal, reason=reason
    )
```

```python
"""Generate the colored-cube sort scene as MJCF.

Geometry follows gear_sonic's g1_loco_manip.xml (two cubes, two pads) extended
to three colors, with open boxes instead of flat pads so that "inside" is a
genuine containment volume.
"""

from dataclasses import dataclass

import numpy as np

from dsrl.config import EnvConfig

COLORS: tuple[str, ...] = ("red", "green", "blue")
COLOR_RGBA: dict[str, str] = {
    "red": "0.9 0.1 0.1 1",
    "green": "0.1 0.9 0.1 1",
    "blue": "0.1 0.1 0.9 1",
}

CUBE_HALF = 0.035
TARGET_HALF_XY = 0.09
TARGET_WALL = 0.005
TARGET_HEIGHT = 0.05
TABLE_TOP_Z = 0.75
CUBE_REGION = ((0.35, 0.60), (-0.25, 0.25))  # x range, y range
TARGET_REGION = ((0.62, 0.78), (-0.35, 0.35))
MIN_CUBE_SEPARATION = 0.10


@dataclass(frozen=True)
class Layout:
    cube_xy: tuple[tuple[float, float], ...]
    target_xy: tuple[tuple[float, float], ...]
    colors: tuple[str, ...]
    preplaced: int


def _rejection_sample(rng, region, count, min_sep) -> tuple[tuple[float, float], ...]:
    (x_lo, x_hi), (y_lo, y_hi) = region
    points: list[tuple[float, float]] = []
    for _ in range(10_000):
        if len(points) == count:
            break
        candidate = (float(rng.uniform(x_lo, x_hi)), float(rng.uniform(y_lo, y_hi)))
        if all(
            ((candidate[0] - p[0]) ** 2 + (candidate[1] - p[1]) ** 2) ** 0.5 > min_sep
            for p in points
        ):
            points.append(candidate)
    if len(points) != count:
        raise RuntimeError(f"could not place {count} objects in {region}")
    return tuple(points)


def sample_layout(rng: np.random.Generator, cfg: EnvConfig) -> Layout:
    n = cfg.n_cubes
    colors = tuple(rng.permutation(np.array(COLORS[:n], dtype=object)).tolist())
    return Layout(
        cube_xy=_rejection_sample(rng, CUBE_REGION, n, MIN_CUBE_SEPARATION),
        target_xy=_rejection_sample(rng, TARGET_REGION, n, 2 * TARGET_HALF_XY + 0.02),
        colors=colors,
        preplaced=int(rng.choice(np.array(cfg.preplaced_choices))),
    )


def _target_body(color: str, x: float, y: float) -> str:
    rgba = COLOR_RGBA[color]
    h, w, t = TARGET_HALF_XY, TARGET_WALL, TARGET_HEIGHT
    return f"""
    <body name="target_{color}" pos="{x} {y} {TABLE_TOP_Z}">
      <geom name="target_{color}_floor" type="box" pos="0 0 {w}"
            size="{h} {h} {w}" rgba="{rgba}"/>
      <geom name="target_{color}_wx1" type="box" pos="{h} 0 {t / 2}"
            size="{w} {h} {t / 2}" rgba="{rgba}"/>
      <geom name="target_{color}_wx2" type="box" pos="{-h} 0 {t / 2}"
            size="{w} {h} {t / 2}" rgba="{rgba}"/>
      <geom name="target_{color}_wy1" type="box" pos="0 {h} {t / 2}"
            size="{h} {w} {t / 2}" rgba="{rgba}"/>
      <geom name="target_{color}_wy2" type="box" pos="0 {-h} {t / 2}"
            size="{h} {w} {t / 2}" rgba="{rgba}"/>
    </body>"""


def _cube_body(color: str, x: float, y: float) -> str:
    return f"""
    <body name="cube_{color}" pos="{x} {y} {TABLE_TOP_Z + CUBE_HALF + 0.002}">
      <freejoint name="cube_{color}_joint"/>
      <geom name="cube_{color}_geom" type="box" size="{CUBE_HALF} {CUBE_HALF} {CUBE_HALF}"
            rgba="{COLOR_RGBA[color]}" mass="0.1" friction="1.0"
            solimp="0.998 0.998 0.001"/>
    </body>"""


def build_scene_xml(layout: Layout, robot_include: str | None) -> str:
    """Emit the full MJCF. `robot_include` is None in unit tests (objects only)."""
    include = f'  <include file="{robot_include}"/>\n' if robot_include else ""
    bodies = "".join(
        _cube_body(color, *layout.cube_xy[i]) + _target_body(color, *layout.target_xy[i])
        for i, color in enumerate(layout.colors)
    )
    return f"""<mujoco model="dsrl_color_sort">
{include}  <option timestep="0.002"/>
  <worldbody>
    <light pos="0 0 3" dir="0 0 -1"/>
    <geom name="floor" type="plane" size="5 5 0.05" rgba="0.3 0.4 0.5 1"/>
    <body name="table_body" pos="0.6 0 {TABLE_TOP_Z - 0.02}">
      <geom name="table_top" type="box" size="0.5 0.6 0.02" rgba="0.6 0.3 0.1 1"/>
      <geom name="table_leg1" type="box" pos="0 0.5 {-TABLE_TOP_Z / 2}"
            size="0.05 0.05 {TABLE_TOP_Z / 2}" rgba="0.6 0.3 0.1 1"/>
      <geom name="table_leg2" type="box" pos="0 -0.5 {-TABLE_TOP_Z / 2}"
            size="0.05 0.05 {TABLE_TOP_Z / 2}" rgba="0.6 0.3 0.1 1"/>
    </body>{bodies}
  </worldbody>
</mujoco>
"""


def write_scene(path: str, layout: Layout, robot_include: str | None) -> None:
    with open(path, "w") as handle:
        handle.write(build_scene_xml(layout, robot_include))
```

```python
"""Fixed-capacity chunk-level replay buffer.

Stores in fp16 by default: at ~30-40 KB per transition in fp32, 100k
transitions would be 3-4 GB, and halving that matters more than the precision.
"""

from typing import NamedTuple

import numpy as np


class Transition(NamedTuple):
    obs_actor: np.ndarray
    obs_critic: np.ndarray
    noise: np.ndarray
    chunk: np.ndarray
    reward: np.ndarray
    done: np.ndarray
    next_obs_actor: np.ndarray
    next_obs_critic: np.ndarray


class ReplayBuffer:
    def __init__(
        self,
        capacity: int,
        obs_actor_dim: int,
        obs_critic_dim: int,
        noise_dim: int,
        chunk_dim: int,
        dtype=np.float16,
    ) -> None:
        self.capacity = capacity
        self._size = 0
        self._cursor = 0
        z = lambda dim: np.zeros((capacity, dim), dtype=dtype)  # noqa: E731
        self._obs_actor = z(obs_actor_dim)
        self._obs_critic = z(obs_critic_dim)
        self._noise = z(noise_dim)
        self._chunk = z(chunk_dim)
        self._next_obs_actor = z(obs_actor_dim)
        self._next_obs_critic = z(obs_critic_dim)
        self._reward = np.zeros(capacity, dtype=np.float32)
        self._done = np.zeros(capacity, dtype=np.float32)

    def __len__(self) -> int:
        return self._size

    def add(
        self,
        obs_actor: np.ndarray,
        obs_critic: np.ndarray,
        noise: np.ndarray,
        chunk: np.ndarray,
        reward: float,
        done: bool,
        next_obs_actor: np.ndarray,
        next_obs_critic: np.ndarray,
    ) -> None:
        i = self._cursor
        self._obs_actor[i] = obs_actor
        self._obs_critic[i] = obs_critic
        self._noise[i] = noise
        self._chunk[i] = chunk
        self._next_obs_actor[i] = next_obs_actor
        self._next_obs_critic[i] = next_obs_critic
        self._reward[i] = reward
        self._done[i] = float(done)
        self._cursor = (i + 1) % self.capacity
        self._size = min(self._size + 1, self.capacity)

    def sample(self, batch_size: int, rng: np.random.Generator) -> Transition:
        if self._size == 0:
            raise ValueError("cannot sample from an empty buffer")
        idx = rng.integers(0, self._size, size=batch_size)
        f32 = lambda a: a[idx].astype(np.float32)  # noqa: E731
        return Transition(
            obs_actor=f32(self._obs_actor),
            obs_critic=f32(self._obs_critic),
            noise=f32(self._noise),
            chunk=f32(self._chunk),
            reward=self._reward[idx],
            done=self._done[idx],
            next_obs_actor=f32(self._next_obs_actor),
            next_obs_critic=f32(self._next_obs_critic),
        )
```

```python
"""Actor and critic for noise-space SAC.

The actor is tanh-squashed and scaled to +-noise_scale. GR00T was trained
expecting N(0, I) seeds; an unbounded actor walks to ||w|| >> 1 where the flow
ODE emits garbage that the critic has never scored.
"""

import math

import torch
import torch.nn.functional as F
from torch import nn

LOG_STD_MIN, LOG_STD_MAX = -5.0, 2.0


def _mlp(in_dim: int, hidden: tuple[int, ...], out_dim: int, layernorm: bool) -> nn.Sequential:
    layers: list[nn.Module] = []
    prev = in_dim
    for width in hidden:
        layers.append(nn.Linear(prev, width))
        if layernorm:
            layers.append(nn.LayerNorm(width))
        layers.append(nn.ReLU())
        prev = width
    layers.append(nn.Linear(prev, out_dim))
    return nn.Sequential(*layers)


class Critic(nn.Module):
    """Q(obs, action) -> scalar. LayerNorm is required for high UTD stability."""

    def __init__(self, in_dim: int, hidden: tuple[int, ...]) -> None:
        super().__init__()
        self.net = _mlp(in_dim, hidden, 1, layernorm=True)

    def forward(self, obs: torch.Tensor, action: torch.Tensor) -> torch.Tensor:
        return self.net(torch.cat([obs, action], dim=-1)).squeeze(-1)


class GaussianNoiseActor(nn.Module):
    def __init__(
        self,
        obs_dim: int,
        noise_dim: int,
        hidden: tuple[int, ...],
        noise_scale: float,
    ) -> None:
        super().__init__()
        self.noise_dim = noise_dim
        self.noise_scale = noise_scale
        self.net = _mlp(obs_dim, hidden, 2 * noise_dim, layernorm=False)

    def forward(self, obs: torch.Tensor) -> tuple[torch.Tensor, torch.Tensor]:
        mean, log_std = self.net(obs).chunk(2, dim=-1)
        return mean, log_std.clamp(LOG_STD_MIN, LOG_STD_MAX)

    def _correction(self, u: torch.Tensor) -> torch.Tensor:
        """Numerically stable log|d tanh/du| summed over dims, plus scale term."""
        tanh_term = (2 * (math.log(2.0) - u - F.softplus(-2 * u))).sum(-1)
        return tanh_term + self.noise_dim * math.log(self.noise_scale)

    def log_prob_of_pre_squash(self, obs: torch.Tensor, u: torch.Tensor) -> torch.Tensor:
        mean, log_std = self(obs)
        base = torch.distributions.Normal(mean, log_std.exp()).log_prob(u).sum(-1)
        return base - self._correction(u)

    def sample(self, obs: torch.Tensor) -> tuple[torch.Tensor, torch.Tensor]:
        mean, log_std = self(obs)
        dist = torch.distributions.Normal(mean, log_std.exp())
        u = dist.rsample()
        noise = self.noise_scale * torch.tanh(u)
        log_prob = dist.log_prob(u).sum(-1) - self._correction(u)
        return noise, log_prob

    @torch.no_grad()
    def act_deterministic(self, obs: torch.Tensor) -> torch.Tensor:
        mean, _ = self(obs)
        return self.noise_scale * torch.tanh(mean)
```

```python
"""DSRL-NA: noise-aliased latent-space SAC over a frozen diffusion policy.

The bootstrap is routed through Q^w rather than Q^a. That is what keeps the
base policy out of the update path entirely: Q^w consumes noise directly, so
the value at s' needs no flow-ODE evaluation, and Q^a is only ever queried at
actions already in the replay buffer.
"""

import copy

import numpy as np
import torch
import torch.nn.functional as F

from dsrl.config import SacConfig
from dsrl.rl.nets import Critic, GaussianNoiseActor
from dsrl.rl.replay import Transition


class DsrlNaAgent:
    def __init__(
        self,
        obs_actor_dim: int,
        obs_critic_dim: int,
        noise_dim: int,
        chunk_dim: int,
        cfg: SacConfig,
        device: torch.device,
    ) -> None:
        self.cfg = cfg
        self.device = device
        self.noise_dim = noise_dim

        self.actor = GaussianNoiseActor(
            obs_actor_dim, noise_dim, cfg.hidden, cfg.noise_scale
        ).to(device)

        # Noise-space critics: the only networks that are bootstrapped from.
        self.qw1 = Critic(obs_critic_dim + noise_dim, cfg.hidden).to(device)
        self.qw2 = Critic(obs_critic_dim + noise_dim, cfg.hidden).to(device)
        self.qw1_target = copy.deepcopy(self.qw1)
        self.qw2_target = copy.deepcopy(self.qw2)
        for p in list(self.qw1_target.parameters()) + list(self.qw2_target.parameters()):
            p.requires_grad_(False)

        # Action-space critics: carry the TD anchor. Absent in vanilla mode.
        if cfg.use_noise_aliasing:
            self.qa1 = Critic(obs_critic_dim + chunk_dim, cfg.hidden).to(device)
            self.qa2 = Critic(obs_critic_dim + chunk_dim, cfg.hidden).to(device)
            qa_params = list(self.qa1.parameters()) + list(self.qa2.parameters())
            self.opt_qa = torch.optim.Adam(qa_params, lr=cfg.lr)
        else:
            self.qa1 = self.qa2 = None
            self.opt_qa = None

        self.opt_actor = torch.optim.Adam(self.actor.parameters(), lr=cfg.lr)
        self.opt_qw = torch.optim.Adam(
            list(self.qw1.parameters()) + list(self.qw2.parameters()), lr=cfg.lr
        )
        self.log_alpha = torch.zeros(1, requires_grad=True, device=device)
        self.opt_alpha = torch.optim.Adam([self.log_alpha], lr=cfg.lr)
        self.target_entropy = -float(noise_dim)

    @property
    def alpha(self) -> float:
        return float(self.log_alpha.exp().item())

    def _to_torch(self, array: np.ndarray) -> torch.Tensor:
        return torch.as_tensor(array, dtype=torch.float32, device=self.device)

    @torch.no_grad()
    def compute_target(self, batch: Transition) -> torch.Tensor:
        """r + gamma*(1-d)*(min Q^w_target(s', w') - alpha*log pi(w'|s'))."""
        next_obs_a = self._to_torch(batch.next_obs_actor)
        next_obs_c = self._to_torch(batch.next_obs_critic)
        reward = self._to_torch(batch.reward)
        done = self._to_torch(batch.done)

        next_noise, next_log_prob = self.actor.sample(next_obs_a)
        q_next = torch.min(
            self.qw1_target(next_obs_c, next_noise),
            self.qw2_target(next_obs_c, next_noise),
        )
        soft = q_next - self.log_alpha.exp() * next_log_prob
        return reward + self.cfg.gamma * (1.0 - done) * soft

    def update(self, batch: Transition) -> dict[str, float]:
        obs_a = self._to_torch(batch.obs_actor)
        obs_c = self._to_torch(batch.obs_critic)
        noise = self._to_torch(batch.noise)
        chunk = self._to_torch(batch.chunk)

        target = self.compute_target(batch)

        # --- Q^a: TD on the stored chunk (NA mode only) ---
        loss_qa_value = 0.0
        if self.cfg.use_noise_aliasing:
            qa1, qa2 = self.qa1(obs_c, chunk), self.qa2(obs_c, chunk)
            loss_qa = F.mse_loss(qa1, target) + F.mse_loss(qa2, target)
            self.opt_qa.zero_grad(set_to_none=True)
            loss_qa.backward()
            self.opt_qa.step()
            loss_qa_value = float(loss_qa.item())
            with torch.no_grad():
                qw_target_value = torch.min(
                    self.qa1(obs_c, chunk), self.qa2(obs_c, chunk)
                )
        else:
            # Vanilla DSRL: Q^w is itself the TD learner.
            qw_target_value = target

        # --- Q^w: distillation from Q^a (NA) or direct TD (vanilla) ---
        loss_qw = F.mse_loss(self.qw1(obs_c, noise), qw_target_value) + F.mse_loss(
            self.qw2(obs_c, noise), qw_target_value
        )
        self.opt_qw.zero_grad(set_to_none=True)
        loss_qw.backward()
        self.opt_qw.step()

        # --- actor: maximise min Q^w minus entropy cost ---
        new_noise, log_prob = self.actor.sample(obs_a)
        q_pi = torch.min(self.qw1(obs_c, new_noise), self.qw2(obs_c, new_noise))
        loss_actor = (self.log_alpha.exp().detach() * log_prob - q_pi).mean()
        self.opt_actor.zero_grad(set_to_none=True)
        loss_actor.backward()
        self.opt_actor.step()

        # --- temperature ---
        loss_alpha = -(
            self.log_alpha.exp() * (log_prob.detach() + self.target_entropy)
        ).mean()
        self.opt_alpha.zero_grad(set_to_none=True)
        loss_alpha.backward()
        self.opt_alpha.step()

        self._soft_update()
        return {
            "loss_qa": loss_qa_value,
            "loss_qw": float(loss_qw.item()),
            "loss_actor": float(loss_actor.item()),
            "loss_alpha": float(loss_alpha.item()),
            "alpha": self.alpha,
            "q_mean": float(q_pi.mean().item()),
        }

    @torch.no_grad()
    def _soft_update(self) -> None:
        tau = self.cfg.tau
        for online, target in ((self.qw1, self.qw1_target), (self.qw2, self.qw2_target)):
            for p, pt in zip(online.parameters(), target.parameters()):
                pt.mul_(1.0 - tau).add_(tau * p)

    def select_noise(self, obs_actor: np.ndarray, deterministic: bool = False) -> np.ndarray:
        obs = self._to_torch(obs_actor)
        with torch.no_grad():
            if deterministic:
                noise = self.actor.act_deterministic(obs)
            else:
                noise, _ = self.actor.sample(obs)
        return noise.cpu().numpy()
```

```python
"""A synthetic stand-in for a frozen diffusion policy.

The base policy is a fixed random network mapping (state, noise) -> action;
reward is highest when the action matches a state-dependent target. Only some
noises produce a good action, so a noise-space RL agent has something real to
learn, with no GPU and no GR00T.
"""

import numpy as np


class ToyBasePolicy:
    """Frozen, deterministic, nonlinear map from (state, noise) to action."""

    def __init__(self, state_dim: int, noise_dim: int, action_dim: int, seed: int = 0) -> None:
        rng = np.random.default_rng(seed)
        hidden = 32
        self.w1 = rng.standard_normal((state_dim + noise_dim, hidden)) / np.sqrt(hidden)
        self.w2 = rng.standard_normal((hidden, action_dim)) / np.sqrt(hidden)

    def __call__(self, state: np.ndarray, noise: np.ndarray) -> np.ndarray:
        x = np.concatenate([state, noise], axis=-1)
        return np.tanh(np.tanh(x @ self.w1) @ self.w2)


class ToyNoiseEnv:
    """One-step episodes: reward = -||base(state, noise) - target(state)||^2."""

    def __init__(self, base_policy: ToyBasePolicy, state_dim: int, action_dim: int, seed: int = 0):
        self.base = base_policy
        self.state_dim = state_dim
        self.action_dim = action_dim
        self.rng = np.random.default_rng(seed)
        self._state = np.zeros(state_dim, dtype=np.float32)

    def _target(self, state: np.ndarray) -> np.ndarray:
        return np.tanh(state[: self.action_dim] * 2.0)

    def reset(self) -> np.ndarray:
        self._state = self.rng.standard_normal(self.state_dim).astype(np.float32)
        return self._state

    def step(self, noise: np.ndarray) -> tuple[np.ndarray, float, bool, np.ndarray]:
        action = self.base(self._state, noise).astype(np.float32)
        reward = -float(np.sum((action - self._target(self._state)) ** 2))
        return self.reset(), reward, True, action
```

```python
"""Map a compact SAC action onto GR00T's full (horizon, max_action_dim) seed.

The raw seed is 40 x 132 = 5280 values. Handing that to SAC is hopeless, and
most of it is padding the model never turns into robot motion. We let the actor
control only the live action dims of the executed prefix and fill everything
else with a fixed per-run draw: in-distribution in magnitude, deterministic in
value, so it adds no unmodelled randomness.
"""

import numpy as np

from dsrl.config import GrootConfig, SacConfig


def noise_dim(groot: GrootConfig, sac: SacConfig, execute_steps: int) -> int:
    steps = execute_steps if sac.restrict_to_prefix else groot.action_horizon
    return steps * groot.live_action_dim


def make_fixed_tail(groot: GrootConfig, seed: int) -> np.ndarray:
    """One standard-normal draw per run, reused for every non-actor entry."""
    rng = np.random.default_rng(seed)
    return rng.standard_normal(
        (groot.action_horizon, groot.max_action_dim)
    ).astype(np.float32)


def assemble_seed(
    actor_noise: np.ndarray,
    fixed_tail: np.ndarray,
    groot: GrootConfig,
    sac: SacConfig,
    execute_steps: int,
) -> np.ndarray:
    """Scatter the actor's compact output into the model's full seed tensor."""
    expected = noise_dim(groot, sac, execute_steps)
    if actor_noise.shape[-1] != expected:
        raise ValueError(
            f"actor noise width {actor_noise.shape[-1]} != expected {expected}"
        )
    batch = actor_noise.shape[0]
    steps = execute_steps if sac.restrict_to_prefix else groot.action_horizon

    seed = np.broadcast_to(
        fixed_tail, (batch, groot.action_horizon, groot.max_action_dim)
    ).copy()
    seed[:, :steps, : groot.live_action_dim] = actor_noise.reshape(
        batch, steps, groot.live_action_dim
    )
    return seed
```

```python
"""MuJoCo sort environment.

Implements the three stubs gear_sonic leaves empty (update_reward,
get_privileged_obs, reset) and overrides check_fall, which upstream calls
self.reset() from inside itself -- convenient for teleop, corrupting for RL.

Drives MuJoCo directly so tests do not require the gear_sonic install; the
closed-loop runner (Task 12) supplies the WBC and cameras on top.
"""

import numpy as np

import mujoco

from dsrl.config import DsrlConfig
from dsrl.sim.reward import advance_settle_counts, is_contained, is_settled
from dsrl.sim.scene import CUBE_HALF, Layout, build_scene_xml
from dsrl.sim.types import CubeState, SortState, TargetState


class SortEnv:
    def __init__(self, layout: Layout, cfg: DsrlConfig, robot_include: str | None = None) -> None:
        self.cfg = cfg
        self.robot_include = robot_include
        self.layout = layout
        self.fell = False
        self._chunk_index = 0
        self._settle_counts: tuple[int, ...] = ()
        self._build(layout)

    # --- construction ---------------------------------------------------

    def _build(self, layout: Layout) -> None:
        self.layout = layout
        xml = build_scene_xml(layout, self.robot_include)
        self.mj_model = mujoco.MjModel.from_xml_string(xml)
        self.mj_data = mujoco.MjData(self.mj_model)
        self._colors = tuple(layout.colors)
        self._cube_bodies = {c: self.mj_model.body(f"cube_{c}").id for c in self._colors}
        self._cube_joints = {
            c: self.mj_model.joint(f"cube_{c}_joint") for c in self._colors
        }
        self._target_bodies = {c: self.mj_model.body(f"target_{c}").id for c in self._colors}

    # --- state ----------------------------------------------------------

    def _targets(self) -> tuple[TargetState, ...]:
        from dsrl.sim.scene import TARGET_HALF_XY, TARGET_HEIGHT

        out = []
        for color in self._colors:
            body = self._target_bodies[color]
            pos = tuple(float(v) for v in self.mj_data.xpos[body])
            rot = self.mj_data.xmat[body].reshape(3, 3)
            out.append(
                TargetState(
                    color=color,
                    center=(pos[0], pos[1], pos[2] + TARGET_HEIGHT / 2),
                    half_extent=(TARGET_HALF_XY, TARGET_HALF_XY, TARGET_HEIGHT),
                    up_axis=tuple(float(v) for v in rot[:, 2]),
                )
            )
        return tuple(out)

    def _cubes(self) -> tuple[CubeState, ...]:
        out = []
        for color in self._colors:
            body = self._cube_bodies[color]
            pos = tuple(float(v) for v in self.mj_data.xpos[body])
            adr = self._cube_joints[color].dofadr[0]
            vel = self.mj_data.qvel[adr : adr + 3]
            out.append(
                CubeState(
                    position=pos,
                    color=color,
                    speed=float(np.linalg.norm(vel)),
                    grasped=self._is_grasped(color),
                )
            )
        return tuple(out)

    def _is_grasped(self, color: str) -> bool:
        """Contact-based: cube touching a non-table, non-target geom."""
        cube_geom = self.mj_model.geom(f"cube_{color}_geom").id
        for i in range(self.mj_data.ncon):
            contact = self.mj_data.contact[i]
            pair = (contact.geom1, contact.geom2)
            if cube_geom not in pair:
                continue
            other = pair[0] if pair[1] == cube_geom else pair[1]
            name = mujoco.mj_id2name(self.mj_model, mujoco.mjtObj.mjOBJ_GEOM, other) or ""
            if not name.startswith(("table", "target", "cube", "floor")):
                return True
        return False

    def observe(self) -> SortState:
        cubes = self._cubes()
        targets = self._targets()
        placed = tuple(
            (t := next((x for x in targets if x.color == c.color), None)) is not None
            and is_contained(c, t)
            and is_settled(self._settle_counts[i], self.cfg.reward)
            for i, c in enumerate(cubes)
        )
        return SortState(
            cubes=cubes,
            targets=targets,
            active_index=self._active_index(cubes, targets),
            settle_counts=self._settle_counts,
            placed_mask=placed,
            pelvis_height=self._pelvis_height(),
            chunk_index=self._chunk_index,
        )

    def _active_index(self, cubes, targets) -> int:
        for i, cube in enumerate(cubes):
            target = next((t for t in targets if t.color == cube.color), None)
            correct = target is not None and is_contained(cube, target)
            if not (correct and is_settled(self._settle_counts[i], self.cfg.reward)):
                return i
        return 0

    def _pelvis_height(self) -> float:
        if self.robot_include is None:
            return 0.75  # no robot in unit-test scenes
        return float(self.mj_data.body("pelvis").xpos[2])

    # --- dynamics -------------------------------------------------------

    def sim_step(self) -> None:
        mujoco.mj_step(self.mj_model, self.mj_data)
        state = SortState(
            cubes=self._cubes(),
            targets=self._targets(),
            active_index=0,
            settle_counts=self._settle_counts,
            placed_mask=(False,) * len(self._colors),
            pelvis_height=self._pelvis_height(),
            chunk_index=self._chunk_index,
        )
        self._settle_counts = advance_settle_counts(state, self.cfg.reward)

    def check_fall(self) -> bool:
        """Report only. Upstream resets here; that corrupts RL transitions."""
        self.fell = self._pelvis_height() < self.cfg.reward.fall_height
        return self.fell

    def set_cube_pose(self, color: str, xyz: tuple[float, float, float]) -> None:
        joint = self._cube_joints[color]
        adr = joint.qposadr[0]
        self.mj_data.qpos[adr : adr + 3] = xyz
        self.mj_data.qpos[adr + 3 : adr + 7] = (1.0, 0.0, 0.0, 0.0)
        dof = joint.dofadr[0]
        self.mj_data.qvel[dof : dof + 6] = 0.0
        mujoco.mj_forward(self.mj_model, self.mj_data)

    def reset(self, layout: Layout | None = None) -> SortState:
        if layout is not None:
            self._build(layout)
        mujoco.mj_resetData(self.mj_model, self.mj_data)
        self._settle_counts = (0,) * len(self._colors)
        self._chunk_index = 0
        self.fell = False
        mujoco.mj_forward(self.mj_model, self.mj_data)

        # Pre-place the first `preplaced` cubes in their matching targets so
        # short episodes still cover late-task states.
        for color in self._colors[: self.layout.preplaced]:
            target = next(t for t in self._targets() if t.color == color)
            self.set_cube_pose(color, (target.center[0], target.center[1], target.center[2]))
        for _ in range(200):
            self.sim_step()
        self._settle_counts = tuple(
            self.cfg.reward.settle_steps if i < self.layout.preplaced else 0
            for i in range(len(self._colors))
        )
        return self.observe()

    # --- RL plumbing ----------------------------------------------------

    def get_privileged_obs(self) -> np.ndarray:
        state = self.observe()
        parts: list[float] = []
        for cube in state.cubes:
            parts.extend(cube.position)
            parts.append(cube.speed)
            parts.append(float(cube.grasped))
        for target in state.targets:
            parts.extend(target.center)
            parts.extend(target.up_axis)
        parts.extend(float(f) for f in state.placed_mask)
        onehot = [0.0] * len(state.cubes)
        onehot[state.active_index] = 1.0
        parts.extend(onehot)
        parts.append(state.pelvis_height)
        return np.asarray(parts, dtype=np.float32)
```

```python
"""Split GR00T's forward pass so the actor can choose the seed.

The backbone pass (expensive) runs first; its pooled features are the actor's
observation, and the actor's seed is consumed by the flow ODE afterwards. The
observation is therefore free -- an intermediate the model already computes.
"""

import numpy as np
import torch

from dsrl.config import GrootConfig, SacConfig
from dsrl.groot.masking import assemble_seed, make_fixed_tail


class SteerableGr00t:
    def __init__(
        self,
        policy,
        groot_cfg: GrootConfig,
        sac_cfg: SacConfig,
        execute_steps: int,
        device: torch.device,
        tail_seed: int = 0,
    ) -> None:
        self.policy = policy
        self.model = policy.model
        self.head = self.model.action_head
        self.groot_cfg = groot_cfg
        self.sac_cfg = sac_cfg
        self.execute_steps = execute_steps
        self.device = device
        self.fixed_tail = make_fixed_tail(groot_cfg, tail_seed)

    @torch.no_grad()
    def encode(self, observation: dict):
        """Run the backbone once; return (features_bundle, pooled actor obs)."""
        batch = self.policy.apply_transforms(observation)
        backbone_input, action_input = self.model.prepare_input(batch)
        backbone_output = self.model.backbone(backbone_input)
        features = self.head._encode_features(backbone_output, action_input)

        pooled_vl = features.backbone_features.float().mean(dim=1)
        pooled_state = features.state_features.float().mean(dim=1)
        obs_actor = torch.cat([pooled_vl, pooled_state], dim=-1).cpu().numpy()
        bundle = (features, backbone_output, action_input)
        return bundle, obs_actor

    @torch.no_grad()
    def decode(self, bundle, actor_noise: np.ndarray) -> np.ndarray:
        """Integrate the flow ODE from the actor's chosen seed."""
        features, backbone_output, action_input = bundle
        seed = assemble_seed(
            actor_noise, self.fixed_tail, self.groot_cfg, self.sac_cfg, self.execute_steps
        )
        seed_t = torch.as_tensor(seed, device=self.device)
        out = self.head.get_action_with_features(
            backbone_features=features.backbone_features,
            state_features=features.state_features,
            embodiment_id=action_input.embodiment_id,
            backbone_output=backbone_output,
            action_input=action_input,
            options={"dsrl_noise": seed_t},
        )
        return out["action_pred"].float().cpu().numpy()
```

- [ ] **Step 2: Write `dsrl/env/chunk_env.py`**

```python
"""Chunk-level RL environment.

One step(noise) == one GR00T decision == `execute_steps` MuJoCo steps. This is
the only module where GR00T, the WBC, and MuJoCo compose; it owns the temporal
abstraction and all resets.

RTC is deliberately off: we never place an `action` key in the observation, so
the flow ODE integrates purely from our injected seed. See spec 5.3 -- with RTC
on, the first rtc_overlap_steps of the seed would be silently overwritten.
"""

import numpy as np

from dsrl.config import DsrlConfig
from dsrl.groot.masking import noise_dim
from dsrl.sim.reward import advance_settle_counts, compute_reward
from dsrl.sim.scene import sample_layout


class ChunkEnv:
    def __init__(self, sort_env, steerable, wbc, cfg: DsrlConfig, seed: int = 0) -> None:
        self.sort_env = sort_env
        self.steerable = steerable
        self.wbc = wbc
        self.cfg = cfg
        self.rng = np.random.default_rng(seed)
        self.noise_dim = noise_dim(cfg.groot, cfg.sac, cfg.env.execute_steps)
        self._prev_chunk: np.ndarray | None = None
        self._chunk_index = 0

    def sample_random_noise(self) -> np.ndarray:
        """Baseline / warm-up action: standard normal clipped to the actor bound."""
        raw = self.rng.standard_normal((1, self.noise_dim)).astype(np.float32)
        return np.clip(raw, -self.cfg.sac.noise_scale, self.cfg.sac.noise_scale)

    def _observe(self):
        observation = self.wbc.build_observation(self.sort_env)
        bundle, obs_actor = self.steerable.encode(observation)
        privileged = self.sort_env.get_privileged_obs()
        obs_critic = np.concatenate([obs_actor[0], privileged])[None, :]
        return bundle, obs_actor, obs_critic

    def reset(self, seed: int | None = None):
        if seed is not None:
            self.rng = np.random.default_rng(seed)
        layout = sample_layout(self.rng, self.cfg.env)
        self.sort_env.reset(layout)
        self.wbc.reset()
        self._prev_chunk = None
        self._chunk_index = 0
        self._bundle, obs_actor, obs_critic = self._observe()
        return obs_actor, obs_critic

    def step(self, actor_noise: np.ndarray):
        chunk = self.steerable.decode(self._bundle, actor_noise)  # [1, H, max_dim]
        executed = chunk[0, : self.cfg.env.execute_steps, :]

        delta = 0.0
        if self._prev_chunk is not None:
            delta = float(np.linalg.norm(executed - self._prev_chunk))
        self._prev_chunk = executed.copy()

        prev_state = self.sort_env.observe()
        reward = 0.0
        terms: dict[str, float] = {}
        terminal = False
        reason = ""

        for action in executed:
            self.wbc.apply(self.sort_env, action)
            self.sort_env.sim_step()
            self.sort_env.check_fall()
            state = self.sort_env.observe()
            result = compute_reward(state, prev_state, 0.0, self.cfg.reward)
            reward += result.reward
            for key, value in result.terms.items():
                terms[key] = terms.get(key, 0.0) + value
            prev_state = state
            if result.terminal:
                terminal, reason = True, result.reason
                break

        # jerk is charged once per chunk, not once per sim step
        jerk = self.cfg.reward.jerk_coeff * delta
        reward += jerk
        terms["jerk"] = terms.get("jerk", 0.0) + jerk

        self._chunk_index += 1
        self.sort_env._chunk_index = self._chunk_index
        timeout = self._chunk_index >= self.cfg.env.max_chunks
        done = terminal or timeout
        if timeout and not terminal:
            reason = "timeout"

        self._bundle, obs_actor, obs_critic = self._observe()
        info = {"terms": terms, "reason": reason, "chunk_index": self._chunk_index}
        return obs_actor, obs_critic, reward, done, info
```

- [ ] **Step 3: Write `dsrl/env/vec.py`**

Only proceed with the multi-process form if Step 2's preflight found the WBC to be in-process. If it is a C++ binary behind ZMQ, set `n_workers=1` and revisit after measuring throughput.

```python
"""N sim workers with GR00T batched in the parent process.

The sim and WBC are CPU-bound and parallelise across processes; GR00T is
GPU-bound and must not be replicated. Workers therefore own only SortEnv+WBC
and exchange observations/actions with the parent over pipes.

DDS/ZMQ and the real-time rate limiter of gear_sonic's BaseSimulator are
bypassed entirely -- we drive DefaultEnv.sim_step() directly, so the sim runs
at CPU speed rather than wall-clock.
"""

import multiprocessing as mp

import numpy as np


def _worker(conn, make_env, seed):
    env = make_env(seed)
    try:
        while True:
            command, payload = conn.recv()
            if command == "reset":
                conn.send(env.reset(seed=payload))
            elif command == "step":
                conn.send(env.step(payload))
            elif command == "close":
                return
            else:
                raise RuntimeError(f"unknown command {command}")
    finally:
        conn.close()


class VecChunkEnv:
    def __init__(self, make_env, n_workers: int, base_seed: int = 0) -> None:
        ctx = mp.get_context("spawn")
        self.parents, self.procs = [], []
        for i in range(n_workers):
            parent, child = ctx.Pipe()
            proc = ctx.Process(
                target=_worker, args=(child, make_env, base_seed + i), daemon=True
            )
            proc.start()
            child.close()
            self.parents.append(parent)
            self.procs.append(proc)
        self.n_workers = n_workers

    def reset(self, seeds=None):
        seeds = seeds if seeds is not None else [None] * self.n_workers
        for parent, seed in zip(self.parents, seeds):
            parent.send(("reset", seed))
        results = [parent.recv() for parent in self.parents]
        obs_actor = np.concatenate([r[0] for r in results], axis=0)
        obs_critic = np.concatenate([r[1] for r in results], axis=0)
        return obs_actor, obs_critic

    def step(self, noises: np.ndarray):
        for i, parent in enumerate(self.parents):
            parent.send(("step", noises[i : i + 1]))
        results = [parent.recv() for parent in self.parents]
        obs_actor = np.concatenate([r[0] for r in results], axis=0)
        obs_critic = np.concatenate([r[1] for r in results], axis=0)
        rewards = np.array([r[2] for r in results], dtype=np.float32)
        dones = np.array([r[3] for r in results], dtype=np.float32)
        infos = [r[4] for r in results]
        return obs_actor, obs_critic, rewards, dones, infos

    def close(self) -> None:
        for parent in self.parents:
            parent.send(("close", None))
        for proc in self.procs:
            proc.join(timeout=5)
```

```python
"""Shared rollout harness. Used for both the Milestone 1 baseline and eval, so
the two numbers are never produced by divergent code paths.
"""

import json
import time
from dataclasses import asdict, dataclass

import numpy as np


@dataclass
class BaselineReport:
    episodes: int
    per_cube_rate: float
    full_sort_rate: float
    chunks_per_sec: float
    mean_episode_chunks: float
    reason_counts: dict[str, int]


def rollout_episode(env, select_noise, seed):
    """One per-cube episode. `select_noise(obs_actor) -> noise`."""
    obs_actor, obs_critic = env.reset(seed=seed)
    transitions, chunks = [], 0
    while True:
        noise = select_noise(obs_actor)
        next_a, next_c, reward, done, info = env.step(noise)
        transitions.append(
            dict(
                obs_actor=obs_actor[0], obs_critic=obs_critic[0], noise=noise[0],
                chunk=env._prev_chunk.reshape(-1), reward=reward, done=done,
                next_obs_actor=next_a[0], next_obs_critic=next_c[0],
            )
        )
        obs_actor, obs_critic = next_a, next_c
        chunks += 1
        if done:
            return transitions, info["reason"], chunks


def rollout_full_sort(env, select_noise, seed, n_cubes):
    """Chain n_cubes episodes WITHOUT reset -- the headline metric."""
    env.reset(seed=seed)
    for _ in range(n_cubes):
        while True:
            noise = select_noise(env._observe()[1])
            _, _, _, done, info = env.step(noise)
            if done:
                break
        if info["reason"] != "correct_placement":
            return False
        env._chunk_index = 0
    return True


def run_baseline(env, select_noise, seeds, n_cubes) -> BaselineReport:
    start = time.time()
    total_chunks = 0
    per_cube_successes = 0
    reasons: dict[str, int] = {}
    all_transitions = []

    for seed in seeds:
        transitions, reason, chunks = rollout_episode(env, select_noise, seed)
        all_transitions.extend(transitions)
        total_chunks += chunks
        reasons[reason] = reasons.get(reason, 0) + 1
        per_cube_successes += int(reason == "correct_placement")

    full = sum(
        rollout_full_sort(env, select_noise, seed, n_cubes) for seed in seeds
    )
    elapsed = time.time() - start

    return BaselineReport(
        episodes=len(seeds),
        per_cube_rate=per_cube_successes / len(seeds),
        full_sort_rate=full / len(seeds),
        chunks_per_sec=total_chunks / max(elapsed, 1e-6),
        mean_episode_chunks=total_chunks / len(seeds),
        reason_counts=reasons,
    ), all_transitions


def save(report: BaselineReport, transitions, path: str) -> None:
    with open(path, "w") as handle:
        json.dump(asdict(report), handle, indent=2)
    np.savez_compressed(
        path.replace(".json", "_transitions.npz"),
        **{k: np.stack([t[k] for t in transitions]) for k in transitions[0]},
    )
```

```python
"""DSRL-NA training loop.

Warm-starts the replay buffer from the Milestone 1 baseline rollouts, so the
measurement that gated the project doubles as offline pretraining data for Q^a
(RLPD-style offline/online mix).
"""

import json
from pathlib import Path

import numpy as np
import torch

from dsrl.config import DsrlConfig
from dsrl.eval.rollout import rollout_full_sort
from dsrl.rl.agent import DsrlNaAgent
from dsrl.rl.replay import ReplayBuffer


def load_warm_start(buffer: ReplayBuffer, path: Path) -> int:
    """Seed the buffer with baseline transitions. Returns the count added."""
    if not path.exists():
        return 0
    data = np.load(path)
    count = len(data["reward"])
    for i in range(count):
        buffer.add(
            obs_actor=data["obs_actor"][i],
            obs_critic=data["obs_critic"][i],
            noise=data["noise"][i],
            chunk=data["chunk"][i],
            reward=float(data["reward"][i]),
            done=bool(data["done"][i]),
            next_obs_actor=data["next_obs_actor"][i],
            next_obs_critic=data["next_obs_critic"][i],
        )
    return count


def train(
    cfg: DsrlConfig,
    vec_env,
    eval_env,
    obs_actor_dim: int,
    obs_critic_dim: int,
    noise_dim: int,
    chunk_dim: int,
    total_env_steps: int,
    eval_every: int,
    eval_seeds: list[int],
    out_dir: Path,
    warm_start: Path | None = None,
    device: str = "cuda",
) -> None:
    out_dir = Path(out_dir)
    out_dir.mkdir(parents=True, exist_ok=True)
    metrics_file = out_dir / "metrics.jsonl"
    eval_file = out_dir / "eval.jsonl"

    rng = np.random.default_rng(0)
    agent = DsrlNaAgent(
        obs_actor_dim=obs_actor_dim,
        obs_critic_dim=obs_critic_dim,
        noise_dim=noise_dim,
        chunk_dim=chunk_dim,
        cfg=cfg.sac,
        device=torch.device(device),
    )
    buffer = ReplayBuffer(
        cfg.sac.replay_capacity, obs_actor_dim, obs_critic_dim, noise_dim, chunk_dim
    )

    if warm_start is not None:
        added = load_warm_start(buffer, warm_start)
        print(f"warm start: {added} baseline transitions loaded")

    obs_actor, obs_critic = vec_env.reset()
    n = vec_env.n_workers

    for step in range(total_env_steps):
        if len(buffer) < cfg.sac.warmup_transitions:
            noises = np.clip(
                rng.standard_normal((n, noise_dim)), -cfg.sac.noise_scale, cfg.sac.noise_scale
            ).astype(np.float32)
        else:
            noises = agent.select_noise(obs_actor)

        next_a, next_c, rewards, dones, infos = vec_env.step(noises)
        for i in range(n):
            buffer.add(
                obs_actor=obs_actor[i], obs_critic=obs_critic[i], noise=noises[i],
                chunk=infos[i]["chunk_flat"], reward=float(rewards[i]),
                done=bool(dones[i]), next_obs_actor=next_a[i], next_obs_critic=next_c[i],
            )
        obs_actor, obs_critic = next_a, next_c

        record = {"step": step, "reward_mean": float(rewards.mean())}
        if len(buffer) >= cfg.sac.warmup_transitions:
            for _ in range(cfg.sac.utd):
                record.update(agent.update(buffer.sample(cfg.sac.batch_size, rng)))

        # Log the per-term breakdown, not just the scalar: a rising return with
        # a growing `disturb` term means the picker is learning to cheat.
        terms: dict[str, float] = {}
        for info in infos:
            for key, value in info["terms"].items():
                terms[key] = terms.get(key, 0.0) + value / n
        record["terms"] = terms
        with metrics_file.open("a") as handle:
            handle.write(json.dumps(record) + "\n")

        if step % eval_every == 0 and step > 0:
            successes = sum(
                rollout_full_sort(
                    eval_env,
                    lambda o: agent.select_noise(o, deterministic=True),
                    seed,
                    cfg.env.n_cubes,
                )
                for seed in eval_seeds
            )
            entry = {
                "step": step,
                "full_sort_rate": successes / len(eval_seeds),
                "n_eval_seeds": len(eval_seeds),
            }
            with eval_file.open("a") as handle:
                handle.write(json.dumps(entry) + "\n")
            print(f"[{step}] full_sort_rate={entry['full_sort_rate']:.3f}")
```

`ChunkEnv.step` must add `info["chunk_flat"] = executed.reshape(-1)` so the buffer receives the executed chunk. Add that line in Task 13 Step 2's `info` dict, and assert `chunk_dim == execute_steps * max_action_dim` at construction.

**Eval uses the same held-out `eval_seeds` as the Milestone 1 baseline.** Comparing against a baseline measured on different layouts is the easiest way to produce a meaningless result.
